Контракты типов и членов volan-dialect-api для 1.0.0. Это сигнатуры для справки, не самостоятельная программа. `internal`/`private constructor` недоступен приложению; protected-члены предназначены для подклассов.

**Типы:** `SqlRenderer`

```kotlin
public abstract class SqlRenderer(

    override fun render(select: SqlSelect): SqlStatement

    override fun render(delete: SqlDelete): SqlStatement

    override fun quote(name: String): String

    protected open val likeEscape: Char get()

    protected open val caseInsensitiveLikeKeyword: String get()

    protected open fun Builder.appendSelect(select: SqlSelect)

    protected open fun Builder.appendDistinct(select: SqlSelect): Unit

    protected fun Builder.appendStandardDistinct(select: SqlSelect)

    protected open fun Builder.appendItems(items: List<SqlSelectItem>)

    protected open fun Builder.appendOrderBy(terms: List<SqlOrder>)

    protected open fun Builder.appendNulls(term: SqlOrder)

    protected open fun Builder.appendLimit(limit: @Nullable Int?, offset: @Nullable Int?): Unit

    protected fun Builder.appendStandardLimit(limit: @Nullable Int?, offset: @Nullable Int?)

    protected open fun Builder.appendInsert(insert: SqlInsert): Unit

    protected fun Builder.appendStandardInsert(insert: SqlInsert)

    protected open fun Builder.appendUpdate(update: SqlUpdate)

    protected open fun Builder.appendDelete(delete: SqlDelete)

    protected open fun Builder.appendReturning(columns: List<String>)

    protected open val defaultValuesClause: String get()

    protected open fun Builder.appendCondition(condition: SqlCondition)

    protected open fun Builder.appendCompare(condition: SqlCondition.Compare)

    protected open fun Builder.appendTextMatch(condition: SqlCondition.TextMatch): Unit

    protected fun Builder.appendStandardTextMatch(condition: SqlCondition.TextMatch)

    protected open fun Builder.appendInList(condition: SqlCondition.InList)

    protected open fun Builder.appendGroup(conditions: List<SqlCondition>, keyword: String, emptyMatches: Boolean)

    protected open fun Builder.appendTuple(condition: SqlCondition.CompareTuple)

    protected open fun Builder.appendExpression(expression: SqlExpression)

    protected fun qualified(table: String, alias: @Nullable String?): String

    protected fun pattern(match: SqlTextMatch, value: String): String

    protected class Builder

        public fun append(text: String): Builder

        public fun append(character: Char): Builder

        public fun append(number: Int): Builder

        public fun bind(value: @Nullable Any?): Builder
```
