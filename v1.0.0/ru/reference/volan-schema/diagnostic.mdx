    OPTIONAL_LIST_TYPE("E0109"),

    /** A type was given more than one list marker, as in `Post[][]`. */
    NESTED_LIST_TYPE("E0110"),

    /** A type was given more than one optional marker, as in `String??`. */
    REPEATED_OPTIONAL_MARKER("E0111"),
