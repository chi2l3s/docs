Type/member contracts of volan-runtime for 1.0.0. These signatures are reference material, not a standalone program. Internal/private constructors are not callable by applications; protected members serve subclasses.

**Types:** `RelationSlot`, `Companion`, `SelectedFields`, `Companion`, `VolanConfigurationException`, `VolanConnectionException`, `VolanConstraintException`, `VolanFieldNotSelectedException`, `VolanForeignKeyException`, `VolanMappingException`, `VolanNotFoundException`, `VolanQueryException`, `VolanRelationNotLoadedException`, `VolanTimeoutException`, `VolanTransactionException`, `VolanUniqueConstraintException`, `VolanUnsupportedException`, `VolanValidationException`

public class VolanRelationNotLoadedException internal constructor(public val model: String, public val relation: String) :
    VolanException(
        "`$model.$relation` was not loaded by the query that produced this row.\n" +
            "  Add it to the query:  include { $relation { } }\n" +
            "  Or check first:       row.is${relation.replaceFirstChar { it.uppercase() }}Loaded",
    )

public class VolanFieldNotSelectedException internal constructor(
    public val model: String,
    public val field: String,
    block: String,
) : VolanException(
    "`$model.$field` was not selected by the query that produced this row.\n" +
        "  Add it to the query:  $block { $field }\n" +
        "  Or check first:       row.is${field.replaceFirstChar { it.uppercase() }}Selected",
)

public class VolanNotFoundException(public val model: String, message: String) : VolanException(message)
